cmake_minimum_required(VERSION 3.24)
project(PULP VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(PULP_BUILD_APP "Build the PULP program (fetches JUCE)" ON)
option(PULP_ASAN "Build engine, render and tests with AddressSanitizer" OFF)

if(MSVC)
  add_compile_options(/utf-8 /permissive- /Zc:preprocessor)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
  # Release builds keep their symbols in a .pdb beside the build, so a crash log names functions.
  # /PDBALTPATH records only the .pdb's file name in the program, not the folder it was built in.
  add_compile_options($<$<CONFIG:Release>:/Zi>)
  add_link_options($<$<CONFIG:Release>:/DEBUG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF> $<$<CONFIG:Release>:/INCREMENTAL:NO>
                   $<$<CONFIG:Release>:/PDBALTPATH:%_PDB%>)
  if(PULP_ASAN)
    add_compile_options(/fsanitize=address /Zi)
    add_link_options(/INCREMENTAL:NO)
  endif()
endif()

# ---------------------------------------------------------------- engine
# Everything that makes sound. No JUCE, no window, no audio device, no OS calls.
file(GLOB_RECURSE PULP_ENGINE_SRC CONFIGURE_DEPENDS
     "${CMAKE_SOURCE_DIR}/engine/src/*.cpp"
     "${CMAKE_SOURCE_DIR}/engine/src/*.h"
     "${CMAKE_SOURCE_DIR}/engine/include/pulp/*.h")
add_library(pulp_engine STATIC ${PULP_ENGINE_SRC})
target_include_directories(pulp_engine PUBLIC "${CMAKE_SOURCE_DIR}/engine/include")
if(MSVC)
  target_compile_options(pulp_engine PRIVATE /W4 /wd4100 /wd4244 /wd4267 /wd4305 /wd4456 /wd4458 /wd4459 /fp:precise)
endif()

# ---------------------------------------------------------------- render
file(GLOB PULP_RENDER_SRC CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/render/*.cpp")
add_executable(pulp_render ${PULP_RENDER_SRC})
target_link_libraries(pulp_render PRIVATE pulp_engine)

# The synthesized part of the factory library is generated by the renderer.
add_custom_target(factory
  COMMAND pulp_render --make-factory "${CMAKE_SOURCE_DIR}/factory"
  DEPENDS pulp_render
  COMMENT "Generating the synthesized factory sounds")

# ---------------------------------------------------------------- tests
file(GLOB PULP_TESTS_SRC CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/tests/*.cpp" "${CMAKE_SOURCE_DIR}/tests/*.h")
add_executable(pulp_tests ${PULP_TESTS_SRC})
target_link_libraries(pulp_tests PRIVATE pulp_engine)
target_compile_definitions(pulp_tests PRIVATE PULP_SOURCE_DIR="${CMAKE_SOURCE_DIR}")

# ---------------------------------------------------------------- app
if(PULP_BUILD_APP)
  include(FetchContent)
  set(FETCHCONTENT_BASE_DIR "${CMAKE_SOURCE_DIR}/.deps" CACHE PATH "" FORCE)
  set(FETCHCONTENT_QUIET OFF)
  # JUCE is pinned to release tag 9.0.4 by URL and hash.
  FetchContent_Declare(JUCE
    URL https://github.com/juce-framework/JUCE/archive/refs/tags/9.0.4.zip
    URL_HASH SHA256=16ad4cfca70b73df6352e4e840ba5dc2e755d36b8ab2dd976ef3ecedcd18f6a9
    DOWNLOAD_EXTRACT_TIMESTAMP ON)
  FetchContent_MakeAvailable(JUCE)

  file(GLOB_RECURSE PULP_APP_SRC CONFIGURE_DEPENDS
       "${CMAKE_SOURCE_DIR}/app/*.cpp" "${CMAKE_SOURCE_DIR}/app/*.h")
  juce_add_gui_app(pulp_app
    PRODUCT_NAME "PULP"
    COMPANY_NAME "PULP"
    VERSION ${PROJECT_VERSION}
    NEEDS_CURL FALSE
    NEEDS_WEB_BROWSER FALSE)
  target_sources(pulp_app PRIVATE ${PULP_APP_SRC})
  # NoDAW design: bundled typefaces (SIL Open Font License, see app/fonts/*-OFL.txt)
  juce_add_binary_data(pulp_fonts SOURCES
    "${CMAKE_SOURCE_DIR}/app/fonts/Inter-Regular.ttf"
    "${CMAKE_SOURCE_DIR}/app/fonts/Inter-SemiBold.ttf"
    "${CMAKE_SOURCE_DIR}/app/fonts/JetBrainsMono-Medium.ttf")
  target_include_directories(pulp_app PRIVATE "${CMAKE_SOURCE_DIR}/app")
  target_compile_definitions(pulp_app PRIVATE
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_ASIO=1
    JUCE_WASAPI=1
    JUCE_WASAPI_EXCLUSIVE=1
    JUCE_DIRECTSOUND=0
    JUCE_PLUGINHOST_VST3=1
    JUCE_REPORT_APP_USAGE=0
    JUCE_STRICT_REFCOUNTEDPOINTER=1
    JUCE_APPLICATION_NAME_STRING="$<TARGET_PROPERTY:pulp_app,JUCE_PRODUCT_NAME>"
    JUCE_APPLICATION_VERSION_STRING="$<TARGET_PROPERTY:pulp_app,JUCE_VERSION>")
  target_link_libraries(pulp_app PRIVATE
    pulp_engine
    pulp_fonts
    juce::juce_gui_basics
    juce::juce_audio_devices
    juce::juce_audio_processors
    juce::juce_recommended_config_flags)

  # Two small VST3 plugins for checking the plugin hosting (build.bat testplugins); never part of dist.
  add_subdirectory(tools/testplugins "${CMAKE_BINARY_DIR}/testplugins" EXCLUDE_FROM_ALL)

  # dist: a folder that runs on its own when PULP.exe is double-clicked.
  add_custom_target(dist
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_SOURCE_DIR}/dist/PULP"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:pulp_app>" "${CMAKE_SOURCE_DIR}/dist/PULP/PULP.exe"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:pulp_render>" "${CMAKE_SOURCE_DIR}/dist/PULP/pulp_render.exe"
    COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different "${CMAKE_SOURCE_DIR}/factory" "${CMAKE_SOURCE_DIR}/dist/PULP/factory"
    DEPENDS pulp_app pulp_render factory
    COMMENT "Copying PULP into dist/PULP")
endif()
